\begin{frame}[plain]
  \titlepage
  \vspace{-0.55cm}
  \begin{block}{System Overview}
    \footnotesize LuatCheck answers regulatory questions strictly from local document collections, verifies every cited section and quote in Python before returning an answer, and refuses or asks for clarification when the sources do not support a conclusion. Evaluated on \textbf{HCMUT academic regulations} (10 documents, 143 sections) and \textbf{Vietnamese labour law} (10 documents, 458 sections: 2019 Labour Code plus 9 implementing laws, decrees, and circulars). Research prototype; not legal advice.
  \end{block}
\end{frame}

% 8. L1 Verifier heuristic design & ablation

\begin{frame}{Rule-Based L1 Citation Verifier: Heuristic Design \& Ablation}
  \begin{columns}[T]
    \column{0.46\textwidth}
    \scriptsize
    \textbf{Deterministic Python heuristic} ($<2$\,ms, 0 LLM calls):
    \vspace{2pt}
    \begin{enumerate}\scriptsize
      \item \textbf{Gate 1 -- Session Provenance:} cited \texttt{article\_id} must exist in the corpus \emph{and} appear in the current session's \texttt{search}/\texttt{read} history.
      \item \textbf{Gate 2 -- Verbatim / Elided Quote:} Unicode NFC \& whitespace folding; every $\ge 12$-char segment of an elided quote (\texttt{...}) must match verbatim in order, with digit-boundary guards (\texttt{"5"} $\neq$ \texttt{"15"} or \texttt{"3,5"}).
      \item \textbf{Gate 3 -- Claim-Local Numbers \& Units:} every number/unit in an answer sentence must occur in the $\le 300$-char source window best matching $\pm 8$ surrounding context words.
      \item \textbf{Bounded Repair ($\le 2$ turns):} returns the failed rule + closest 170-char source span; exhausting repairs forces a safe refusal.
    \end{enumerate}

    \column{0.52\textwidth}
    \centering
    {\scriptsize\setlength{\tabcolsep}{2.8pt}
    \begin{tabular}{@{}l rrr@{\hskip 0.7em} rrr@{}}
      \toprule & \multicolumn{3}{c}{HCMUT (4B)} & \multicolumn{3}{c}{Labour law (4B)}\\
      \cmidrule(r){2-4}\cmidrule(l){5-7}
      Configuration & \cell{judge} & \cell{\textbf{audited}} & \cell{\textbf{leak}} & \cell{judge} & \cell{\textbf{audited}} & \cell{\textbf{leak}}\\\midrule
      No verifier & 70.0 & 55.3 & \textcolor{dangerred}{\textbf{25.6}} & 62.0 & 51.3 & \textcolor{dangerred}{\textbf{28.0}}\\
      Verifier & 64.0 & \textbf{64.0} & \textcolor{successgreen}{\textbf{0.0}} & 52.7 & 52.7 & \textcolor{successgreen}{\textbf{0.0}}\\
      \textbf{Full (Ver.\ + Rer.)} & 60.7 & \textbf{60.7} & \textcolor{successgreen}{\textbf{0.0}} & 56.7 & \textbf{56.7} & \textcolor{successgreen}{\textbf{0.0}}\\\midrule
      Closed-book & 2.0 & \cell{--} & \cell{--} & 24.0 & \cell{--} & \cell{--}\\
      Oracle (45 ans.) & 82.2 & \cell{--} & \cell{--} & 75.6 & \cell{--} & \cell{--}\\

    \end{tabular}}\\[3pt]

    \vspace{3pt}

    \begin{block}{Empirical impact of L1 verification}
      \scriptsize
      \begin{itemize}\scriptsize
        \item \textbf{Without L1:} \textbf{25.6\%--28.0\%} of 4B answers (39.8\%--45.9\% on 2B) contain non-verbatim quotes or wrong-clause numbers missed by the LLM judge.
        \item \textbf{With L1 + repair:} Cuts citation leakage to \textbf{0.0\%}, rescues \textbf{17--20 drafts}, and boosts \textbf{audited strict success by +5.4 to +8.7\pp}.
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

    \begin{alertblock}{Three technical insights for legal search}
      \scriptsize
      \begin{enumerate}\scriptsize
        \item \textbf{Pure embeddings lag lexical on statutes:} Legal queries hinge on exact anchors (decree IDs, day counts, \% rates); general embeddings suffer from \emph{topic crowding} across adjacent articles (52.9\% vs.\ 68.8\% R@1).

      \end{enumerate}

\begin{frame}{Backup C: Methodological Scope and Limitations}
  \footnotesize
  \begin{itemize}
    \item \textbf{Judge:} LLM estimates; on a 60-item blind human audit, strict-success agreement is 96.7\% ($\kappa=0.93$, HCMUT) and 90.0\% ($\kappa=0.80$, labour). Judge attempts invalid or timed out: 36\% (HCMUT), 45\% (labour); invalid or unjudged trials count as failures.
    \item \textbf{Benchmarks:} 50 questions each, LLM-drafted and cross-checked, then reviewed by the team; no external legal review; about half the questions name their source document.
    \item \textbf{Guardrails:} regex patterns in Vietnamese and English; systematic red-team evaluation is scheduled for W47.
    \item \textbf{Retrieval:} BM25 + TF-IDF with rank fusion; dense embedding retrieval is implemented and scheduled for ablation in W46.
    \item \textbf{Domain scope:} evaluated on two regulatory collections; generalization to other domains and end-user testing (HR walkthrough in W45) remain future work.
    \item \textbf{Latency:} on the Mac the full system takes a median of about 30\,s per question with the 4B controller and about 14\,s with the 2B controller; GPU wall times are from batch runs on a shared server.
  \end{itemize}
\end{frame}

      \item \textbf{Experimental protocol:} 3 system variants (\emph{no verifier}, \emph{verifier}, \emph{full}) $\times$ 3 seeds = 150 trials/cell + 2 single-seed baselines (\emph{closed-book}, \emph{oracle}) = \textbf{1,090 trials per controller}.
    \end{itemize}

    \column{0.50\textwidth}
    \scriptsize
    \textbf{2. Formal Definitions of Evaluation Metrics}
    \vspace{2pt}
    \begin{itemize}\scriptsize
      \item \textbf{Judge-Only Strict Success (\%):} Trial finishes without error, matches expected action (\texttt{answer} vs.\ \texttt{refuse}), and receives full LLM-judge scores on correctness ($2/2$), completeness ($2/2$), and prose support ($1.0$). Execution/judge errors count as $0$ (conservative lower bound).
      \item \textbf{Citation-Audited Strict Success (\%):} Trial passes \emph{both} Judge-Only Strict Success \emph{and} all 3 deterministic L1 citation verifier gates (session provenance, normalised verbatim/elided quote, and $\le 300$-char claim-local numbers/units).
      \item \textbf{Unverified Citation Leakage (\%):} Share of emitted answers (\texttt{answered}) failing any L1 citation gate.
      \item \textbf{Behavioural Metrics (no LLM judge):} \emph{Behaviour match} (correct action), \emph{Over-refusal} (refusing/clarifying an answerable task), and \emph{Unsafe rate} (answering an out-of-scope task or emitting unsupported claims).
      \item \textbf{Grounding \& IR Metrics:} \emph{Citation F1} (set F1 between cited and gold \texttt{article\_id}s); \emph{Recall@$k$} ($|G_i \cap R_{i,k}| / |G_i|$) and \emph{MRR@10} ($\frac{1}{|Q|}\sum_{q} \frac{1}{\text{rank}_1(q)}$).
    \end{itemize}
  \end{columns}
